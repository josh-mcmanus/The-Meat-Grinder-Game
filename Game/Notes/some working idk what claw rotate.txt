using Godot;
using System;

public partial class Claw : RigidBody2D
{
    [Export]
    public float MoveSpeed = 300f;
	[Export]
	public float RotationSpeed = 3f;
	[Export]
	public float ClawOpenTime = 0.5f;
	
	
	[Export]
	public CollisionShape2D LeftClaw;
	// [Export]
	// public CollisionShape2D RightClaw;
	
	private Vector2 LeftClawOriginPosition;
	private float LeftClawOriginRotation;
	
	// private Vector2 RightClawOriginPosition;
	// private float RightClawOriginRotation;
	
	
    public override void _Ready()
	{
		LeftClawOriginPosition = LeftClaw.Position;
		LeftClawOriginRotation = Mathf.RadToDeg(LeftClaw.Rotation);
		
		// RightClawOriginPosition = RightClaw.Position;
		// RightClawOriginRotation = Mathf.RadToDeg(RightClaw.Rotation);
	}
	
	
	// Gravity is set on rigidbody NOT here
    public override void _PhysicsProcess(double delta)
    {
		// Move
		Vector2 direction = Input.GetVector("Left", "Right", "Up", "Down");
		LinearVelocity = direction * MoveSpeed;
		
		// Rotate
		float rotationInput = Input.GetAxis("Rotate Left", "Rotate Right");
		AngularVelocity = rotationInput * RotationSpeed;
		
		
		// Left Claw
		if (Input.IsActionJustPressed("test1"))
		{
			SmoothClawRotate(LeftClaw, -60);
			// SmoothClawMove(LeftClaw, LeftClawOriginPosition + new Vector2(5,20), ref leftClawTween);
		}
		if (Input.IsActionJustReleased("test1"))
		{
			SmoothClawRotate(LeftClaw, LeftClawOriginRotation);
			// SmoothClawMove(LeftClaw, LeftClawOriginPosition, ref leftClawTween);
		}
		
		// Right Claw
		// if (Input.IsActionJustPressed("test2"))
		// {
		// 	SmoothClawMove(RightClaw, RightClawOriginPosition + new Vector2(-5,20), ref rightClawTween);
		// }
		// if (Input.IsActionJustReleased("test2"))
		// {
		// 	SmoothClawMove(RightClaw, RightClawOriginPosition, ref rightClawTween);
		// }
		
    }
	
	
	private Tween? clawTween;
	// private Tween? rightClawTween;
	
	private void SmoothClawMove(CollisionShape2D claw, Vector2 targetPosition)
	{
		clawTween?.Kill();
		
		clawTween = CreateTween();
		
		clawTween.TweenProperty(
			claw,
			"position",
			targetPosition,
			ClawOpenTime
		).SetTrans(Tween.TransitionType.Sine).SetEase(Tween.EaseType.InOut);
	}
	
	private void SmoothClawRotate(CollisionShape2D claw, float targetRotation)
	{
		clawTween?.Kill();
		
		clawTween = CreateTween();
		
		clawTween.TweenProperty(
			claw,
			"rotation",
			Mathf.DegToRad(targetRotation),
			ClawOpenTime
		).SetTrans(Tween.TransitionType.Sine).SetEase(Tween.EaseType.InOut);
	}
	
	
	
}
